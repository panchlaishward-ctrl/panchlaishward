async function performSearch() { const nameQuery = document.getElementById("nameInput").value.toLowerCase().trim(); const dobQuery = document.getElementById("dobInput").value.trim(); const resultsDiv = document.getElementById("results"); const statusDiv = document.getElementById("status"); resultsDiv.innerHTML = ""; if (!nameQuery && !dobQuery) { statusDiv.innerText = "অনুগ্রহ করে নাম অথবা জন্ম তারিখের যেকোনো একটি লিখুন।"; return; } statusDiv.innerText = "পিডিএফ ফাইলসমূহে খোঁজা হচ্ছে, অনুগ্রহ করে অপেক্ষা করুন..."; let found = false; const dobEn = convertDigits(dobQuery, false).toLowerCase(); const dobBn = convertDigits(dobQuery, true).toLowerCase(); for (const pdfInfo of pdfList) { if (found) break; try { const loadingTask = pdfjsLib.getDocument(pdfInfo.url); const pdf = await loadingTask.promise; for (let pageNum = 1; pageNum <= pdf.numPages; pageNum++) { if (found) break; const page = await pdf.getPage(pageNum); const textContent = await page.getTextContent(); const pageText = textContent.items.map(item => item.str).join(" "); const personBlocks = pageText.split(/(?=নাম[:\s])/i); // ঠিক ১টি নির্দিষ্ট ব্লকের সাথে মিল খোজা const matchedBlock = personBlocks.find(block => { const lowerBlock = block.toLowerCase(); const matchesName = nameQuery ? lowerBlock.includes(nameQuery) : true; const matchesDob = dobQuery ? (lowerBlock.includes(dobEn) || lowerBlock.includes(dobBn)) : true; if (nameQuery && dobQuery) return matchesName && matchesDob; if (nameQuery) return matchesName; if (dobQuery) return matchesDob; return false; }); if (matchedBlock) { found = true; // ১টি পাওয়ার সাথে সাথেই পুরো লুপ থামিয়ে দেওয়া const name = getValueAfter(matchedBlock, "নাম"); const voterNo = getValueAfter(matchedBlock, "ভোটার নং"); const dob = getValueAfter(matchedBlock, "জন্ম তারিখ"); const fatherOrHusband = matchedBlock.includes("স্বামী") ? getValueAfter(matchedBlock, "স্বামী") : getValueAfter(matchedBlock, "পিতা"); const fatherOrHusbandLabel = matchedBlock.includes("স্বামী") ? "স্বামী" : "পিতা"; const mother = getValueAfter(matchedBlock, "মাতা"); const area = getValueAfter(matchedBlock, "ঠিকানা"); resultsDiv.innerHTML = `
আলী ইসলাম
উদ্যোক্তা, ০৩ নং পাঁচলাইশ ওয়ার্ড
নাম: ${name}
ভোটার নং- ${voterNo}
জন্ম তারিখ: ${dob}
${fatherOrHusbandLabel}: ${fatherOrHusband}
মাতা: ${mother}
এলাকা: ${area}
মূল পিডিএফ ফাইল দেখুন (পৃষ্ঠা ${pageNum})
`; } } } catch (error) { console.error("Error reading PDF:", error); } } statusDiv.innerText = found ? "১টি নির্দিষ্ট তথ্য পাওয়া গেছে:" : "কোনো তথ্য পাওয়া যায়নি।"; }
